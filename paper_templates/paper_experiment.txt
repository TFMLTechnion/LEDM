# REPRODUCIBILITY: the experiment case used proximity reweighting ON

# --- inputs ----------------------------------------------------------
position_file     = ../data/experiment/position.dat
kinematics_file   = ../data/experiment/kinematics.dat
particles_dir     = ../data/experiment/particles
particles_pattern = particles_*.dat

# --- output ----------------------------------------------------------
output_dir    = outputs/experiment         # relative to the working directory
case_name     = experiment
output_format = both                       # npz (authoritative) + tecplot .dat

# --- grid / ROI ------------------------------------------------------
dx             = 1.0          # mm  (set to your experiment grid spacing)
roi_mode       = body
roi_pad        = 2.0
max_grid_nodes = 4000000

# --- LE-DM solver ----------------------------------------------------
boundary_constraints = on
kappa              = 100
lambda_c           = 0.05
coverage_ref_count = 1.0
interp_kernel      = wide
sigma_u            = 0.01     # raise toward the measured PTV velocity 1-sigma
minres_tol         = 1e-11   # NOTE: a loose tol (1e-6) reports "converged"
minres_maxit       = 20000   # while div(u) is still ~1e-1; see the run banner's
use_jacobi_precond = on      # div_rms_norm and constraint_div_tol below.

# --- constraint tolerances (checked AFTER the solve, and warned about) ---
constraint_div_tol  = 1e-3   # dimensionless Delta*rms(div u)/U_ref
constraint_body_tol = 1e-3   # relative residual of the no-slip identity rows
warm_start         = true

# --- proximity reweighting (paper experiment case: ON) ---------------
enable_proximity_reweight = true
sigma_gamma               = 0.5   # near-wall down-weighting length [mm]

# --- coordinate / rotation convention --------------------------------
length_unit = mm
time_unit   = s
angle_unit  = rad
euler_seq   = ZYX
handedness  = right
omega_frame = world
